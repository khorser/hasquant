  265 -- |An array of freshly heap-allocated (@DUP@'d) C strings -- the output-side counterpart of
  266 -- 'withStringArray'. Each element is read via 'peekCString' and the whole array (including every
  267 -- individual string) is then released via 'qlFreeStringArray' in one call, mirroring
  268 -- 'peekDoubleArray'\/'peekIntArray'\''s read-then-free shape.
  269 peekCStringArray :: Ptr CUInt -> Ptr (Ptr CString) -> IO [String]
  270 peekCStringArray pl pp = do
  271   l <- peek pl
  272   p <- peek pp
  273   raws <- peekArray (fromIntegral l) p
  274   strs <- mapM peekCString raws
  275   strs <$ qlFreeStringArray l p

  306 -- |Like 'peekIntArray'\''/'peekDoubleArray', but for an array of a C struct rather than a C
  307 -- primitive: reads the length and pointer out-params, walks the array via 'peekArray' (so the
  308 -- element type just needs a 'Storable' instance -- e.g. one built from c2hs @{#get#}@/@{#sizeof#}@
  309 -- hooks), converts every element via @convert@, /then/ hands the whole array to @freeFn@ to
  310 -- release. The conversion must run before the free -- unlike 'peekIntArray'\''/'peekDoubleArray',
  311 -- whose elements are self-contained primitives, a struct element read here may itself own
  312 -- further heap buffers (e.g. a @char*@/@double*@ field) that @convert@ still needs to dereference
  313 -- (via 'peekCString'/'peekArray' etc.); freeing first would leave it reading already-freed
  314 -- memory. @freeFn@ takes the element count because some frees need it (e.g.
  315 -- 'qlFreeAdditionalResults'); one that doesn't can ignore it.
  316 peekStructArray :: Storable a => (a -> IO b) -> (CUInt -> Ptr a -> IO ()) -> Ptr CUInt -> Ptr (Ptr a) -> IO [b]
  317 peekStructArray convert freeFn pl pp = do
  318   l <- peek pl
  319   p <- peek pp
  320   raws <- peekArray (fromIntegral l) p
  321   results <- mapM convert raws
  322   results <$ freeFn l p
  323 
  324 -- |Like 'peekStructArray' but for a @T**@ array of C++-owned pointers to live objects: peeks the
  325 -- length/pointer out-params, converts each raw pointer to a live Haskell value via @peekOne@
  326 -- (installing that value's own finalizer over the pointee), then frees only the array spine via
  327 -- 'qlFreePointerArray' -- not the pointees, whose lifetime @peekOne@ has now taken over.
  328 peekPtrArray :: (Ptr a -> IO b) -> Ptr CUInt -> Ptr (Ptr (Ptr a)) -> IO [b]
  329 peekPtrArray peekOne pl pp = do
  330   l <- peek pl
  331   p <- peek pp
  332   ptrs <- peekArray (fromIntegral l) p
  333   xs <- mapM peekOne ptrs
  334   xs <$ qlFreePointerArray (castPtr p)

  365 dayIsValid :: Day -> Bool

  367   where s = toModifiedJulianDay' x - qlStart

  419 realMatrixFromVector :: Word -> Word -> RealVector -> Either String RealMatrix

  421   | rows * cols == fromIntegral (V.length d) = Right $ RealMatrix rows cols d
  422   | otherwise = Left $ "Data length " ++ show (V.length d)
  423       ++ " does not match dimensions " ++ show rows ++ "x" ++ show cols
