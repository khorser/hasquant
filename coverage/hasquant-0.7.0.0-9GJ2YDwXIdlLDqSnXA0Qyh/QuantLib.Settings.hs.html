   46 -- |Sets the x87 floating-point unit to 64-bit extended precision.
   47 --
   48 -- Call this once at program start on Windows, before any pricing. By the time Haskell
   49 -- code runs there, the x87 unit sits at 53-bit precision, where a binary linked by
   50 -- clang++ rather than ghc has 64-bit; every 80-bit @long double@ operation inside
   51 -- QuantLib is then silently rounded to double. That matters because @boost::math@ promotes @double@
   52 -- arguments to @long double@ by default, so its distributions lose the precision their
   53 -- algorithms assume -- returning quietly less accurate answers, or failing to converge
   54 -- outright (@hestonSLVFDMModel@ throws out of @quantile(non_central_chi_squared)@
   55 -- without this).
   56 --
   57 -- A no-op on every other platform and on non-x86 Windows. The control word is
   58 -- per-thread, so with a threaded runtime call it on each OS thread that prices. Only
   59 -- the x87 unit is touched: Haskell's own 'Double' arithmetic is SSE\/MXCSR and is
   60 -- unaffected either way.
   61 setExtendedPrecision :: IO ()
   62 setExtendedPrecision =
   63   setExtendedPrecision'_ >>
   64   return ()

  196 -- |Best effort: give the finalizers of unreachable QuantLib objects a chance to run, so
  197 -- their C++ destructors fire before, say, 'setEvaluationDate' notifies surviving observers.
  198 --
  199 -- 'performGC' only /schedules/ finalizers, it does not run them synchronously, so this is a
  200 -- nudge rather than a guarantee. Where correctness depends on an object being gone, anchor
  201 -- its dates so staleness cannot arise in the first place.
  202 collectGarbage :: IO ()
  203 collectGarbage = performGC >> performGC
