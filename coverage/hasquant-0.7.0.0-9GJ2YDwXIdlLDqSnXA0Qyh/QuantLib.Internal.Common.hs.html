 1183 -- |Wrap a Haskell @price -> value@ function as a real QuantLib @Payoff@, usable anywhere a
 1184 -- 'Payoff' is (@QuantLib.Instrument.Option.oneAssetOption@, @multiAssetOption@,
 1185 -- @QuantLib.Instrument.Swap.varianceOption@, @QuantLib.Method.fdmLogInnerValue@,
 1186 -- @fdmCellAveragingInnerValue@, ...) -- the fully custom counterpart to the concrete
 1187 -- pre-implemented payoffs listed by the 'Payoff' constructors above.
 1188 --
 1189 -- The payoff is valid only inside the continuation, and the continuation must span the whole
 1190 -- /use/, not just the construction: every consumer stores the payoff and calls back into it
 1191 -- later (an @Instrument@ at @NPV@ time, an @FdmInnerValueCalculator@ at @fdmSolve@ time), so
 1192 -- pricing must happen before this function returns. Same lifetime rule, and the same reason, as
 1193 -- @QuantLib.Method.withCustomFdmInnerValueCalculator@.
 1194 --
 1195 -- @name@ and @description@ are what QuantLib's own error messages and @Payoff::name@ report; they
 1196 -- are not interpreted.
 1197 --
 1198 -- __Not every engine accepts a non-standard payoff.__ QuantLib's analytic, binomial, finite-
 1199 -- difference and @MCEuropeanEngine@ families all recover the strike by downcasting to
 1200 -- @StrikedTypePayoff@\/@PlainVanillaPayoff@ first, and a further ~30 engines route through
 1201 -- @BlackCalculator@, whose @AcyclicVisitor@ knows only the four built-in striked payoffs. Most of
 1202 -- these fail with a clean QuantLib exception, but
 1203 -- @QuantLib.PricingEngine.fdBlackScholesVanillaEngine@ and
 1204 -- @QuantLib.PricingEngine.fdHestonVanillaEngine@ perform that downcast /unchecked/ upstream and
 1205 -- will __crash the process__, not throw, on a custom payoff. Confirmed-generic consumers:
 1206 -- @QuantLib.Method.fdmLogInnerValue@\/@fdmCellAveragingInnerValue@ (and hence @fdmSolve@), and
 1207 -- @QuantLib.PricingEngine.mcAmericanEngine@ with @controlVariate = False@.

 1209   -> String -- ^description
 1210   -> (Double -> Double) -- ^payoff(price)
 1211   -> (Payoff -> IO b) -> IO b

 1214 -- |As 'withCustomPayoff', but produces a real QuantLib @StrikedTypePayoff@ carrying an
 1215 -- @(optionType, strike)@ pair alongside the Haskell function.
 1216 --
 1217 -- __The pair is advisory: it does not define the payoff__ -- @payoff(price)@ alone does, exactly
 1218 -- as for 'withCustomPayoff'. It exists because QuantLib's finite-difference vanilla engines reach
 1219 -- past the @Payoff@ interface for a strike when sizing their grid:
 1220 -- @FdBlackScholesVanillaEngine@ @dynamic_pointer_cast@s to @StrikedTypePayoff@ /without/ a check
 1221 -- and calls @strike()@ twice -- once for the mesher's extent, once for its node-concentration
 1222 -- point -- then hands the payoff itself to @FdmLogInnerValue@, which takes a plain @Payoff@. So a
 1223 -- payoff built here prices correctly through
 1224 -- 'QuantLib.PricingEngine.fdBlackScholesVanillaEngine' and
 1225 -- 'QuantLib.PricingEngine.fdHestonVanillaEngine', where one built by 'withCustomPayoff' would
 1226 -- crash the process on that unchecked cast. Pass the strike you want the grid centred on.

 1229 -- payoff is valid only inside the continuation, which must span the whole use (pricing included),

 1939 -- `ext::any`, so this Haskell view picks three concrete shapes -- `Real` (`Double`), `std::string`

 1945 -- |Discriminants for `QlAdditionalResult.type`, bound from `enum AdditionalResultType` in

 1948                           | AdditionalResultString

 1986 -- |Registers `struct QlAdditionalResult*` with c2hs as `RawResultPtr`, `nocode` since we supply
 1987 -- the Haskell type ourselves (below) rather than a c2hs-generated wrapper. This is what lets the
 1988 -- `additionalResults` `{#fun#}` binding (in `QuantLib.Instrument`)'s low-level array-of-structs
 1989 -- out-parameter (C type `struct QlAdditionalResult **`) be typed `Ptr RawResultPtr` =
 1990 -- `Ptr (Ptr RawResult)`, instead of defaulting to an opaque `Ptr (Ptr ())`.

 1995 -- |One raw `QlAdditionalResult` entry, peeked field-by-field via c2hs `{#get#}` hooks. Its
 1996 -- `Storable` instance (`sizeOf`/`alignment` from `{#sizeof#}`/`{#alignof#}`, both read straight
 1997 -- from the C struct layout, not hand-computed) is what lets `peekStructArray`
 1998 -- (`QuantLib.Internal`) walk the C array via a plain `peekArray`, rather than hand-rolled pointer
 1999 -- arithmetic.
 2000 data RawResult = RawResult
 2001   { rKey :: CString, rType :: CInt, rDval :: CDouble
 2002   , rSval :: CString, rVarr :: Ptr CDouble, rVlen :: CUInt }
 2003 
 2004 instance Storable RawResult where
 2005   sizeOf _ = 48
 2006 
 2007   alignment _ = 8
 2008 
 2009   peek p = RawResult <$> (\ptr -> do {C2HSImp.peekByteOff ptr 0 :: IO (C2HSImp.Ptr C2HSImp.CChar)}) p
 2010                       <*> (\ptr -> do {C2HSImp.peekByteOff ptr 8 :: IO C2HSImp.CInt}) p
 2011                       <*> (\ptr -> do {C2HSImp.peekByteOff ptr 16 :: IO C2HSImp.CDouble}) p
 2012                       <*> (\ptr -> do {C2HSImp.peekByteOff ptr 24 :: IO (C2HSImp.Ptr C2HSImp.CChar)}) p
 2013                       <*> (\ptr -> do {C2HSImp.peekByteOff ptr 32 :: IO (C2HSImp.Ptr C2HSImp.CDouble)}) p
 2014                       <*> (\ptr -> do {C2HSImp.peekByteOff ptr 40 :: IO C2HSImp.CUInt}) p
 2015   poke = error "RawResult is peek-only (read from C, never constructed in Haskell)"

 2017 -- |Convert one raw entry into its keyed Haskell value. `sval`/`varr` are only read for the
 2018 -- discriminant that owns them; their buffers are released in bulk afterwards, by
 2019 -- `qlFreeAdditionalResults`, not per-field here.
 2020 convertResult :: RawResult -> IO (String, AdditionalResultVal)
 2021 convertResult r = do
 2022   key <- peekCString (rKey r)
 2023   val <- case toEnum (fromIntegral (rType r)) of
 2024     AdditionalResultDouble -> return (RealVal (realToFrac (rDval r)))
 2025     AdditionalResultString -> StringVal <$> peekCString (rSval r)
 2026     AdditionalResultDoubleVector -> RealVectorVal . map realToFrac
 2027                                        <$> peekArray (fromIntegral (rVlen r)) (rVarr r)
 2028     AdditionalResultUnknown -> UnsupportedVal <$> peekCString (rSval r)
 2029   return (key, val)

 2031 -- |Peek the C array of `QlAdditionalResult` into a keyed list, then release the whole array (keys,
 2032 -- `sval`/`varr` buffers, and the array itself) in one `qlFreeAdditionalResults` call.
 2033 peekAdditionalResults :: Ptr CUInt -> Ptr RawResultPtr -> IO [(String, AdditionalResultVal)]
 2034 peekAdditionalResults = peekStructArray convertResult (\l p -> qlFreeAdditionalResults l (castPtr p))
