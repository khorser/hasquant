  709 nextECBDate' a1 a2 =
  710   C2HSImp.withCString a1 $ \a1' -> 
  711   withMaybeDay a2 $ \a2' -> 
  712   preErrorCheck $ \a3' -> 
  713   nextECBDate''_ a1' a2' a3' >>= \res ->
  714   let {res' = toDay res} in
  715   errorCheck  a3'>>
  716   return (res')
