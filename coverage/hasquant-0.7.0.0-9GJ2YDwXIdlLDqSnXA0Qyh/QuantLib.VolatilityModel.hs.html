  432 simpleLocalVolatilityEstimator yearFraction series = do
  433   (ds, vs) <- qlSimpleLocalVolatilityEstimator yearFraction dates vals
  434   return $ zip ds vs
  435   where (dates, vals) = unzip (toList series)
  436 qlSimpleLocalVolatilityEstimator :: (Double) -> ([Day]) -> ([Double]) -> IO (([Day]), ([Double]))
  437 qlSimpleLocalVolatilityEstimator a1 a2 a3 =
  438   let {a1' = realToFrac a1} in 
  439   withDayArray a2 $ \(a2'1, a2'2) -> 
  440   withDoubleArray a3 $ \(a3'1, a3'2) -> 
  441   preArray $ \(a4'1, a4'2) -> 
  442   preArray $ \(a5'1, a5'2) -> 
  443   preErrorCheck $ \a6' -> 
  444   qlSimpleLocalVolatilityEstimator'_ a1' a2'1  a2'2 a3'1  a3'2 a4'1  a4'2 a5'1  a5'2 a6' >>
  445   peekDayArray  a4'1  a4'2>>= \a4'' -> 
  446   peekDoubleArray  a5'1  a5'2>>= \a5'' -> 
  447   errorCheck  a6'>>
  448   return (a4'', a5'')

  453 unzipBars bars = (map d5 xs, map o5 xs, map c5 xs, map h5 xs, map l5 xs)
  454   where
  455     xs = toList bars

  457     o5 (_,o,_,_,_) = o

  459     h5 (_,_,_,h,_) = h
