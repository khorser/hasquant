 1548 -- |QuantLib's own BTP-vs-EUR-swap-curve relative-value tool
 1549 -- (@ql\/instruments\/bonds\/btp.hpp@'s @RendistatoCalculator@): aggregates a 'RendistatoBasket'
 1550 -- into a weighted BTP yield\/duration, prices a fixed ladder of 1..15Y EUR swaps against a
 1551 -- discount curve, and reports the swap whose duration is closest to (without exceeding) the
 1552 -- basket's own duration as the \"equivalent swap\". @euriborForwardCurve@ forwards the internally
 1553 -- constructed Euribor index used for those comparison swaps' floating leg -- matching upstream's
 1554 -- own @Euribor@ default when @Nothing@, but the calculator immediately prices those swaps
 1555 -- (@fairRate@), which needs a real forwarding curve to project floating cashflows, so a
 1556 -- @Nothing@ here throws rather than degrading gracefully; pass the same curve as
 1557 -- @discountCurve@ unless a genuinely different forward curve is wanted. @discountCurve@ is
 1558 -- required, with no upstream default.
 1559 rendistatoCalculator :: (RendistatoBasket) -- ^basket
 1560  -> ((Int,TimeUnit)) -- ^euriborTenor
 1561  -> (Maybe (GenYieldTermStructure y1)) -- ^euriborForwardCurve
 1562  -> (GenYieldTermStructure y2) -- ^discountCurve
 1563  -> IO ((RendistatoCalculator))
 1564 rendistatoCalculator a1 a2 a3 a4 =
 1565   withRendistatoBasket a1 $ \a1' -> 
 1566   let {(a2'1, a2'2) = fromEnumQuantity a2} in 
 1567   withMaybeYieldTermStructure a3 $ \a3' -> 
 1568   withYieldTermStructure a4 $ \a4' -> 
 1569   preErrorCheck $ \a5' -> 
 1570   rendistatoCalculator'_ a1' a2'1  a2'2 a3' a4' a5' >>= \res ->
 1571   peekRendistatoCalculator res >>= \res' ->
 1572   errorCheck  a5'>>
 1573   return (res')
