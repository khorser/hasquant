  820 toMatrixDouble :: (Word, Word, [Double]) -> Matrix Double
  821 toMatrixDouble (r, c, d) = Matrix r c d

  823 -- |Eigenvalues and eigenvectors of a real symmetric matrix, computed by the symmetric threshold
  824 -- Jacobi algorithm. The eigenvalues come back in decreasing order, and the eigenvectors are the
  825 -- /columns/ of the returned matrix: column @i@ belongs to the @i@-th eigenvalue.
  826 symmetricSchurDecomposition :: Matrix Double -- ^symmetric matrix
  827   -> IO ([Double], Matrix Double) -- ^eigenvalues, eigenvectors as columns
  828 symmetricSchurDecomposition (Matrix mr mc md) = do
  829   (values, r, c, vectors) <- qlSymmetricSchurDecomposition mr mc md
  830   pure (values, Matrix r c vectors)
  831 qlSymmetricSchurDecomposition :: (Word) -> (Word) -> ([Double]) -> IO (([Double]), (Word), (Word), ([Double]))
  832 qlSymmetricSchurDecomposition a1 a2 a3 =
  833   let {a1' = fromIntegral a1} in 
  834   let {a2' = fromIntegral a2} in 
  835   withDoubleArrayRaw a3 $ \a3' -> 
  836   preArray $ \(a4'1, a4'2) -> 
  837   prePtr $ \a5' -> 
  838   prePtr $ \a6' -> 
  839   preArray $ \(a7'1, a7'2) -> 
  840   preErrorCheck $ \a8' -> 
  841   qlSymmetricSchurDecomposition'_ a1' a2' a3' a4'1  a4'2 a5' a6' a7'1  a7'2 a8' >>
  842   peekDoubleArray  a4'1  a4'2>>= \a4'' -> 
  843   peekWord  a5'>>= \a5'' -> 
  844   peekWord  a6'>>= \a6'' -> 
  845   peekDoubleArray  a7'1  a7'2>>= \a7'' -> 
  846   errorCheck  a8'>>
  847   return (a4'', a5'', a6'', a7'')

  851 -- |Pseudo square root @S@ of a real symmetric matrix @M@, i.e. the matrix with @S*transpose S == M@.
  852 -- When @M@ is not positive semi-definite the given 'SalvagingAlgorithm' approximates it; with
  853 -- 'SalvagingNone' a non-positive-semi-definite input throws instead.
  854 pseudoSqrt :: Matrix Double -- ^symmetric matrix
  855   -> SalvagingAlgorithm
  856   -> IO (Matrix Double)
  857 pseudoSqrt (Matrix mr mc md) salvaging = toMatrixDouble <$> qlPseudoSqrt mr mc md salvaging
  858 qlPseudoSqrt :: (Word) -> (Word) -> ([Double]) -> (SalvagingAlgorithm) -> IO ((Word), (Word), ([Double]))
  859 qlPseudoSqrt a1 a2 a3 a4 =

  861   let {a2' = fromIntegral a2} in 
  862   withDoubleArrayRaw a3 $ \a3' -> 
  863   let {a4' = fromEnumC a4} in 
  864   prePtr $ \a5' -> 
  865   prePtr $ \a6' -> 
  866   preArray $ \(a7'1, a7'2) ->

  869   peekWord  a5'>>= \a5'' -> 
  870   peekWord  a6'>>= \a6'' ->

  873   return (a5'', a6'', a7'')
