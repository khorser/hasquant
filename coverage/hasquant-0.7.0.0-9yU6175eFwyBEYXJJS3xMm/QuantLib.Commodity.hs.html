  509 -- |Whether a date falls within the interval, optionally excluding either end.
  510 isDateBetween :: DateInterval
  511               -> Day -- ^date
  512               -> Bool -- ^includeFirst
  513               -> Bool -- ^includeLast
  514               -> Bool
  515 isDateBetween (startDate, endDate) date includeFirst includeLast =
  516   (if includeFirst then date >= startDate else date > startDate) &&
  517   (if includeLast then date <= endDate else date < endDate)

  519 -- |The overlap of two date intervals, or 'Nothing' if they don't overlap at all (upstream returns
  520 -- a default-constructed, both-dates-null 'DateInterval' in this case; a 'Maybe' is the natural
  521 -- Haskell equivalent).
  522 intersection :: DateInterval -> DateInterval -> Maybe DateInterval
  523 intersection (s1, e1) (s2, e2)
  524   | (s1 < s2 && e1 < s2) || (s1 > e2 && e1 > e2) = Nothing
  525   | otherwise = Just (max s1 s2, min e1 e2)

  633 qlUnitOfMeasureConversionConvert_ :: (UnitOfMeasureConversion) -> (CommodityType) -> (UnitOfMeasure) -> (Double) -> IO ((Double), (CommodityType), (UnitOfMeasure))
  634 qlUnitOfMeasureConversionConvert_ a1 a2 a3 a4 =
  635   withUnitOfMeasureConversion a1 $ \a1' -> 
  636   withCommodityType a2 $ \a2' -> 
  637   withUnitOfMeasure a3 $ \a3' -> 
  638   let {a4' = realToFrac a4} in 
  639   alloca $ \a5' -> 
  640   alloca $ \a6' -> 
  641   preErrorCheck $ \a7' -> 
  642   qlUnitOfMeasureConversionConvert_'_ a1' a2' a3' a4' a5' a6' a7' >>= \res ->
  643   let {res' = realToFrac res} in
  644   peekCommodityTypePtr  a5'>>= \a5'' -> 
  645   peekUnitOfMeasurePtr  a6'>>= \a6'' -> 
  646   errorCheck  a7'>>
  647   return (res', a5'', a6'')

  651 -- |Apply the conversion factor to a quantity, converting it from the conversion's source to its
  652 -- target unit of measure (or vice versa). Throws if the quantity's unit of measure is on neither
  653 -- side of the conversion.
  654 convertQuantity :: UnitOfMeasureConversion -> Quantity -> IO Quantity
  655 convertQuantity conv (ct, uom, amount) = do
  656   (amount', ct', uom') <- qlUnitOfMeasureConversionConvert_ conv ct uom amount
  657   pure (ct', uom', amount')

  659 -- |Combine two conversions sharing a common unit of measure into a derived conversion between
  660 -- their other two units. Throws if the conversions don't share a common unit.
  661 chainUnitOfMeasureConversion :: (UnitOfMeasureConversion) -> (UnitOfMeasureConversion) -> IO ((UnitOfMeasureConversion))
  662 chainUnitOfMeasureConversion a1 a2 =
  663   withUnitOfMeasureConversion a1 $ \a1' -> 
  664   withUnitOfMeasureConversion a2 $ \a2' -> 
  665   preErrorCheck $ \a3' -> 
  666   chainUnitOfMeasureConversion'_ a1' a2' a3' >>= \res ->
  667   peekUnitOfMeasureConversion res >>= \res' ->
  668   errorCheck  a3'>>
  669   return (res')

  673 -- |Look up a (possibly derived, via triangulation) conversion between two units of measure for a
  674 -- given commodity type. Throws if none can be found. Pre-populated with a set of known petroleum
  675 -- conversion factors even before any 'addUomConversion' call.
  676 lookupUomConversion :: (CommodityType) -> (UnitOfMeasure) -- ^source
  677  -> (UnitOfMeasure) -- ^target
  678  -> (UnitOfMeasureConversionType) -> IO ((UnitOfMeasureConversion))
  679 lookupUomConversion a1 a2 a3 a4 =
  680   withCommodityType a1 $ \a1' -> 
  681   withUnitOfMeasure a2 $ \a2' -> 
  682   withUnitOfMeasure a3 $ \a3' -> 
  683   let {a4' = (fromIntegral . fromEnum) a4} in 
  684   preErrorCheck $ \a5' -> 
  685   lookupUomConversion'_ a1' a2' a3' a4' a5' >>= \res ->
  686   peekUnitOfMeasureConversion res >>= \res' ->
  687   errorCheck  a5'>>
  688   return (res')

  696   withUnitOfMeasureConversion a1 $ \a1' ->
