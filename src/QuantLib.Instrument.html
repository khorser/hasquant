-- domestic currency of the underlying FOR/DOM pair; Inverse contracts do so
-- in the foreign currency; Quanto contracts use a separate quanto currency.

-- funding is discrete at the supplied period.

 -> (Calendar) -> (DayCounter) -> IO ((Instrument))

  let {a1' = fromEnumC a1} in 
  let {a2' = fromEnumC a2} in 
  let {(a3'1, a3'2) = fromEnumQuantity a3} in 
  withCalendar a4 $ \a4' -> 
  withDayCounter a5 $ \a5' -> 
  preErrorCheck $ \a6' -> 
  perpetualFutures'_ a1' a2' a3'1  a3'2 a4' a5' a6' >>= \res ->
  peekInstrument res >>= \res' ->
  errorCheck  a6'>>
  return (res')
